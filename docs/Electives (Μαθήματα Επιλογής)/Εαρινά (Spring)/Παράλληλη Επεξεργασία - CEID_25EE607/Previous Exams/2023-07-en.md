**Exam Question I**

Identify and explain any issues in the following OpenMP code. Propose a solution.

```c
int a[20], s;
#pragma omp parallel num_threads(10)
{
    for (int i = 0; i < 20; i++)
    {
        #pragma omp task
        {
            a[i] = func();
        }
    }

    s = a[0];
    for (int j = 1; j < 20; j++) s += a[j];
}
```
