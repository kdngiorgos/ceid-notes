**7.** Ποια τιμή θα εκτυπωθεί όταν ο παρακάτω κώδικας MPI εκτελεστεί με 4 διεργασίες (mpirun -n 4 <executable>); Εξηγήστε.

**8.** Ποιο μήνυμα θα εκτυπώσει η διεργασία με rank = 2 όταν ο παρακάτω κώδικας MPI εκτελεστεί με 4 διεργασίες;

```c
#include <mpi.h>
#include <stdio.h>
int main(int argc, char** argv) {
  MPI_Init(&argc, &argv);
  int rank, size;
  MPI_Comm_rank(MPI_COMM_WORLD, &rank);
  MPI_Comm_size(MPI_COMM_WORLD, &size);
  int input = rank + 2; // Each process has a value: rank + 2
  int output;

  MPI_Scan(&input, &output, 1, MPI_INT, MPI_PROD, MPI_COMM_WORLD);
  printf("Process %d: output = %d\n", rank, output);

  MPI_Finalize();
  return 0;
}
```

- [ ] a. 6
- [ ] b. 9
- [ ] c. 24
- [ ] d. 41

**15.** Αναφέρετε και διορθώστε πιθανά σφάλματα τον παρακάτω πολυνηματικό κώδικα με εντολές SSE.

```c
int nThreads = 16;
float*data;
posix_memalign((void **)&data, 16, 32*sizeof(float)); // allocate 16-Byte aligned memory for 32 float values
for (int i=0; i<32; ++i) data[i] = drand48();

omp_set_num_threads(nThreads);
#pragma omp parallel
{
  float * const myData = data + 2*omp_get_thread_num();
  __m128 c4 = _mm_set1_ps(M_PI); // = 3.141592…
  __m128 r4 = _mm_load_ps(myData);
  __m128 a4 = _mm_mul_ps(c4, _mm_mul_ps(r4, r4));
  _mm_store_ps(myData, a4);
}
```

```c
__global__ void kernel(int *a) {

}

}
```
