2.2 Δίνεται ο παρακάτω κώδικας MPI: [1,5]

```c
#include <mpi.h>
#include <stdio.h>

int main(int argc, char** argv)
{
    MPI_Init(&argc, &argv);

    int rank, size;
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    double data=(double)rank;
    double total=0.;

    MPI_Bcast(&data, 1, MPI_DOUBLE, 2, MPI_COMM_WORLD);
    MPI_Allreduce(&data, &total, 1, MPI_DOUBLE, MPI_PROD, MPI_COMM_WORLD);

    if (rank == 1) printf("%lf\n", total);

    MPI_Finalize();
    return 0;
}
```

```c
#include <omp.h>
void work(int wid, int i, int j)
{
  // κώδικας με σημαντικό χρόνο εκτέλεσης εδώ
}

int main()
{
  int n = 4096;
  int wid;

  #pragma omp parallel
  for (int i=0; i<n; i++) {
    wid = get_work(i); // thread-safe συνάρτηση, ορισμένη κάπου αλλού
    #pragma omp for
    for (int j=0; j<n; j++) {
      work(wid, i, j);
    }
  }

  return 0;
}
```

Θέμα 4° (2 μονάδες)

Ο παρακάτω σειριακός κώδικας αρχικοποιεί το διάνυσμα A διάστασης N και μετά υπολογίζει την ελάχιστη τιμή του καθώς και τη θέση της τιμής αυτής. Θεωρείστε πως η διάσταση N μπορεί να είναι αυθαίρετα μεγάλη. Παραλληλοποιήστε τον κώδικα με χρήση του μοντέλου MPI.

```c
#include <stdio.h>  // printf
#include <stdlib.h> // malloc

#define N (1024*1024)

extern float initialize(int i); // υλοποιημένη αλλού

int main (int argc, char **argv)
{
  float *A;

  A = (float *) malloc(N*sizeof(double));  // δέσμευση μνήμης Α

  // αρχικοποίηση Α, κάθε τιμή είναι διαφορετική
  for (int i = 0; i < N; i++)
    A[i] = initialize(i);

  // compute min
  float min_val = 1e10;  // αρκετά μεγάλη τιμή για το συγκεκριμένο πρόβλημα
  int min_idx = -1;
  for (int i = 0; i < N; i++) {
     if (A[i] < min_val) {
        min_val = A[i];
        min_idx = i;
     }
  }

  printf("min value=%f at position=%d\n", min_val, min_idx);
  free(A);  // αποδέσμευση μνήμης Α
  return 0;
}
```
