```c
int a ;

{
  int b,c ;
  #pragma omp parallel shared(c) private(b)
  {
    int d ;

    }
  }

  // more code here
  return 0;
}
```

- [ ] b. private
- [ ] c. firstprivate
- [ ] d. Τίποτα από τα παραπάνω

```c
#include <mpi.h>
#include <stdio.h>

int main(int argc, char** argv)
{
    MPI_Init(&argc, &argv);

    int rank, size;
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    double total=0.;
    double local=rank;

    MPI_Allreduce(&local, &total, 1, MPI_DOUBLE, MPI_PROD, MPI_COMM_WORLD);

    MPI_Finalize();
    return 0;
}
```

```c
  #pragma omp parallel
  {
    func();
  }

}
```
