    > Οι λύσεις γράφτηκαν από μοντέλα AI και δεν έχουν ελεγχθεί από άνθρωπο. Η αρίθμηση ακολουθεί το αρχείο θεμάτων.

    ## Θέμα 1Α

    *Αξιοπιστία: δύο ανεξάρτητες λύσεις συμφωνούν · εμπιστοσύνη λύτη: υψηλή*

    Υποθέτω ότι εκτυπώνεται το πλήθος για κάθε θέση του αλφαριθμητικού· αν ζητείται μία φορά ανά διαφορετικό χαρακτήρα, αρκεί να προστεθεί πίνακας `printed`.

    ```c
    #include <stdio.h>
    #include <string.h>

    #define MAX 1000

    int main(void) {
        char s[MAX];
        int freq[256] = {0};
        int i;

        printf("Δώσε αλφαριθμητικό: ");
        if (fgets(s, sizeof(s), stdin) == NULL)
            return 1;
        s[strcspn(s, "\n")] = '\0';

        for (i = 0; s[i] != '\0'; i++)
            freq[(unsigned char)s[i]]++;

        for (i = 0; s[i] != '\0'; i++)
            printf("Ο χαρακτήρας '%c' εμφανίζεται %d φορές.\n",
                   s[i], freq[(unsigned char)s[i]]);

        return 0;
    }
    ```

    ## Θέμα 1Β

    *Αξιοπιστία: δύο ανεξάρτητες λύσεις συμφωνούν · εμπιστοσύνη λύτη: υψηλή*

    Στη C μια συνάρτηση δεν μπορεί να επιστρέψει ολόκληρο πίνακα με τιμή· έτσι χρησιμοποιείται πίνακας-έξοδος `freq` που περνά στη συνάρτηση με διεύθυνση.

    ```c
    #include <stdio.h>
    #include <string.h>

    #define MAX 1000
    #define SYMS 256

    void count_chars(const char s[], int freq[SYMS]) {
        int i;

        for (i = 0; i < SYMS; i++)
            freq[i] = 0;

        for (i = 0; s[i] != '\0'; i++)
            freq[(unsigned char)s[i]]++;
    }

    int main(void) {
        char s[MAX];
        int freq[SYMS];
        int i;

        printf("Δώσε αλφαριθμητικό: ");
        if (fgets(s, sizeof(s), stdin) == NULL)
            return 1;
        s[strcspn(s, "\n")] = '\0';

        count_chars(s, freq);

        for (i = 0; s[i] != '\0'; i++)
            printf("Ο χαρακτήρας '%c' εμφανίζεται %d φορές.\n",
                   s[i], freq[(unsigned char)s[i]]);

        return 0;
    }
    ```

    ## Θέμα 1Γ

    *Αξιοπιστία: δύο ανεξάρτητες λύσεις συμφωνούν · εμπιστοσύνη λύτη: υψηλή*

    Η `main` δηλώνει το αλφαριθμητικό `s` και τον πίνακα συχνοτήτων `freq`. Καλεί την `count_chars(s, freq)`: το `s` περνά ως δείκτης σταθερών χαρακτήρων, δηλαδή ως είσοδος, ενώ το `freq` περνά ως δείκτης σε ακέραιους, δηλαδή ως έξοδος. Επειδή οι πίνακες στη C περνούν με διεύθυνση, η συνάρτηση αρχικοποιεί και αυξάνει τα στοιχεία του `freq` και οι αλλαγές είναι ορατές στη `main`. Η συνάρτηση δεν επιστρέφει τιμή· η πληροφορία “επιστρέφει” μέσω του πίνακα `freq`.

    ## Θέμα 2Α

    *Αξιοπιστία: δύο ανεξάρτητες λύσεις συμφωνούν · εμπιστοσύνη λύτη: υψηλή*

    ```c
    #include <stdio.h>
    #include <string.h>

    #define ROWS 3
    #define COLS 5
    #define NAME_LEN 50
    #define AM_LEN 20

    typedef struct {
        int occupied;
        char name[NAME_LEN];
        char am[AM_LEN];
    } Seat;

    int main(void) {
        Seat room[ROWS][COLS];
        int i, j, n, r, c;

        for (i = 0; i < ROWS; i++) {
            for (j = 0; j < COLS; j++) {
                room[i][j].occupied = 0;
                room[i][j].name[0] = '\0';
                room[i][j].am[0] = '\0';
            }
        }

        printf("Πλήθος φοιτητών προς καταχώρηση: ");
        scanf("%d", &n);

        if (n < 0 || n > ROWS * COLS) {
            printf("Λάθος πλήθος φοιτητών.\n");
            return 1;
        }

        for (i = 0; i < n; i++) {
            char name[NAME_LEN], am[AM_LEN];

            printf("Στοιχεία φοιτητή %d: ονοματεπώνυμο σε μια γραμμή, ΑΜ, γραμμή(1-3), στήλη(1-5):\n", i + 1);
            scanf(" %49[^\n]", name);
            scanf("%19s %d %d", am, &r, &c);

            if (r < 1 || r > ROWS || c < 1 || c > COLS || room[r - 1][c - 1].occupied) {
                printf("Η θέση είναι άκυρη ή ήδη κατειλημμένη. Επανάληψη.\n");
                i--;
                continue;
            }

            room[r - 1][c - 1].occupied = 1;
            strcpy(room[r - 1][c - 1].name, name);
            strcpy(room[r - 1][c - 1].am, am);
        }

        printf("\n--- Καταχωρημένοι φοιτητές ---\n");
        for (i = 0; i < ROWS; i++) {
            for (j = 0; j < COLS; j++) {
                if (room[i][j].occupied) {
                    printf("Θέση (%d,%d): %s, ΑΜ %s\n",
                           i + 1, j + 1, room[i][j].name, room[i][j].am);
                }
            }
        }

        {
            char searchName[NAME_LEN], searchAM[AM_LEN];
            int found = 0;

            printf("\n--- Αναζήτηση ---\n");
            printf("Ονοματεπώνυμο: ");
            scanf(" %49[^\n]", searchName);
            printf("ΑΜ: ");
            scanf("%19s", searchAM);

            for (i = 0; i < ROWS && !found; i++) {
                for (j = 0; j < COLS; j++) {
                    if (room[i][j].occupied &&
                        strcmp(room[i][j].name, searchName) == 0 &&
                        strcmp(room[i][j].am, searchAM) == 0) {
                        printf("Ο φοιτητής %s με ΑΜ %s κάθεται στη γραμμή %d, στήλη %d.\n",
                               searchName, searchAM, i + 1, j + 1);
                        found = 1;
                        break;
                    }
                }
            }

            if (!found)
                printf("Ο φοιτητής δεν βρίσκεται στην αίθουσα.\n");
        }

        return 0;
    }
    ```

    ## Θέμα 2Β

    *Αξιοπιστία: δύο ανεξάρτητες λύσεις συμφωνούν · εμπιστοσύνη λύτη: υψηλή*

    Η `main` οργανώθηκε ως εξής:

    1. Δηλώνει έναν δισδιάστατο πίνακα `room[3][5]` από εγγραφές `Seat`, όπου κάθε θέση περιέχει `occupied`, ονοματεπώνυμο και ΑΜ.
    2. Αρχικοποιεί όλες τις θέσεις ως κενές.
    3. Διαβάζει το πλήθος `n` των φοιτητών που θα καταχωρηθούν.
    4. Για κάθε φοιτητή διαβάζει ονοματεπώνυμο, ΑΜ, γραμμή και στήλη. Ελέγχει αν η θέση είναι έγκυρη και κενή· αν είναι, αποθηκεύει τα στοιχεία και σημειώνει τη θέση ως κατειλημμένη.
    5. Διατρέχει όλο τον πίνακα `room` και εκτυπώνει όσες θέσεις είναι κατειλημμένες.
    6. Διαβάζει ονοματεπώνυμο και ΑΜ προς αναζήτηση.
    7. Διατρέχει τον πίνακα και, αν βρει φοιτητή με ίδιο ονοματεπώνυμο και ίδιο ΑΜ, τυπώνει τη γραμμή και τη στήλη του· αλλιώς τυπώνει μήνυμα ότι δεν βρίσκεται.
    8. Τερματίζει.

    ## Θέμα 2Γ

    *Αξιοπιστία: δύο ανεξάρτητες λύσεις συμφωνούν · εμπιστοσύνη λύτη: υψηλή*

    Δήλωση της συνάρτησης, μετά τον ορισμό του `typedef`:

    ```c
    int find_student(Seat room[ROWS][COLS],
                     const char name[],
                     const char am[],
                     int *row,
                     int *col);
    ```

    Ορισμός:

    ```c
    int find_student(Seat room[ROWS][COLS],
                     const char name[],
                     const char am[],
                     int *row,
                     int *col) {
        int i, j;

        for (i = 0; i < ROWS; i++) {
            for (j = 0; j < COLS; j++) {
                if (room[i][j].occupied &&
                    strcmp(room[i][j].name, name) == 0 &&
                    strcmp(room[i][j].am, am) == 0) {
                    *row = i + 1;
                    *col = j + 1;
                    return 1;
                }
            }
        }

        *row = 0;
        *col = 0;
        return 0;
    }
    ```

    Στη `main` αλλάζει μόνο το τμήμα της αναζήτησης· αφαιρείται η μεταβλητή `found` και οι εμφωλευμένοι βρόχοι αναζήτησης:

    ```c
    {
        char searchName[NAME_LEN], searchAM[AM_LEN];
        int row, col;

        printf("\n--- Αναζήτηση ---\n");
        printf("Ονοματεπώνυμο: ");
        scanf(" %49[^\n]", searchName);
        printf("ΑΜ: ");
        scanf("%19s", searchAM);

        if (find_student(room, searchName, searchAM, &row, &col))
            printf("Ο φοιτητής %s με ΑΜ %s κάθεται στη γραμμή %d, στήλη %d.\n",
                   searchName, searchAM, row, col);
        else
            printf("Ο φοιτητής δεν βρίσκεται στην αίθουσα.\n");
    }
    ```

    ## Θέμα 3

    *Αξιοπιστία: δύο ανεξάρτητες λύσεις συμφωνούν · εμπιστοσύνη λύτη: υψηλή*

    Δήλωση:

    ```c
    double average(const int a[], int n);
    ```

    Ορισμός:

    ```c
    double average(const int a[], int n) {
        long sum = 0;
        int i;

        if (n <= 0)
            return 0.0;

        for (i = 0; i < n; i++)
            sum += a[i];

        return (double)sum / n;
    }
    ```

    Πρόγραμμα επίδειξης:

    ```c
    #include <stdio.h>

    #define MAXN 100

    double average(const int a[], int n);

    double average(const int a[], int n) {
        long sum = 0;
        int i;

        if (n <= 0)
            return 0.0;

        for (i = 0; i < n; i++)
            sum += a[i];

        return (double)sum / n;
    }

    int main(void) {
        int a[MAXN];
        int n, i;

        printf("Πλήθος στοιχείων: ");
        scanf("%d", &n);

        if (n < 0 || n > MAXN) {
            printf("Λάθος πλήθος στοιχείων.\n");
            return 1;
        }

        printf("Δώσε %d ακεραίους:\n", n);
        for (i = 0; i < n; i++)
            scanf("%d", &a[i]);

        printf("Μέσος όρος: %.2f\n", average(a, n));

        return 0;
    }
